cmake_minimum_required(VERSION 3.16)
project(fq_installed_consumer LANGUAGES C CXX)
find_package(futures_quote CONFIG REQUIRED)
add_executable(consumer_shared ../../examples/client.c)
set_target_properties(consumer_shared PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(consumer_shared PRIVATE futures_quote::shared)
add_executable(consumer_static ../../examples/client.cpp)
target_link_libraries(consumer_static PRIVATE futures_quote::static)
enable_testing()
add_test(NAME installed_shared COMMAND consumer_shared ${CMAKE_CURRENT_SOURCE_DIR}/../../config/replay.yaml)
add_test(NAME installed_static COMMAND consumer_static ${CMAKE_CURRENT_SOURCE_DIR}/../../config/replay.yaml)
