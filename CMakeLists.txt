cmake_minimum_required(VERSION 3.16)
project(futures_quote VERSION 0.1.0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT APPLE)
  message(FATAL_ERROR "futures_quote currently supports Linux and macOS")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(FQ_BUILD_TESTS "Build correctness tests" ON)
option(FQ_BUILD_APPS "Build command-line tools" ON)
option(FQ_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
set(FQ_SANITIZERS "address,undefined" CACHE STRING "Sanitizers used when FQ_SANITIZE is ON")
if(FQ_SANITIZE)
  add_compile_options(-fsanitize=${FQ_SANITIZERS} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${FQ_SANITIZERS})
endif()
# Embed private yaml-cpp implementation in both artifacts, avoiding a public
# YAML dependency or missing transitive symbols in the static archive.
file(GLOB FQ_YAML_SOURCES CONFIGURE_DEPENDS third_party/yaml-cpp-0.8.0/src/*.cpp)
add_library(fq_yaml OBJECT ${FQ_YAML_SOURCES})
target_include_directories(fq_yaml PRIVATE third_party/yaml-cpp-0.8.0/include third_party/yaml-cpp-0.8.0/src)
target_compile_definitions(fq_yaml PRIVATE YAML_CPP_STATIC_DEFINE YAML=fq_private_yaml)
set_target_properties(fq_yaml PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_library(fq_core OBJECT src/client.cpp src/config.cpp src/registry.cpp src/cme.cpp src/lme.cpp src/transport.cpp src/pcap.cpp)
target_include_directories(fq_core PRIVATE include src generated/cme third_party/yaml-cpp-0.8.0/include)
target_compile_definitions(fq_core PRIVATE FQ_BUILD YAML_CPP_STATIC_DEFINE YAML=fq_private_yaml _GNU_SOURCE)
target_compile_options(fq_core PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(fq_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
foreach(kind IN ITEMS shared static)
  string(TOUPPER ${kind} KIND)
  add_library(futures_quote_${kind} ${KIND} $<TARGET_OBJECTS:fq_core> $<TARGET_OBJECTS:fq_yaml>)
  target_include_directories(futures_quote_${kind} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(futures_quote_${kind} PUBLIC Threads::Threads)
  target_compile_features(futures_quote_${kind} INTERFACE cxx_std_11)
  set_target_properties(futures_quote_${kind} PROPERTIES OUTPUT_NAME futures_quote VERSION ${PROJECT_VERSION} SOVERSION 1 EXPORT_NAME ${kind})
  add_library(futures_quote::${kind} ALIAS futures_quote_${kind})
endforeach()
if(UNIX AND NOT APPLE)
  target_link_options(futures_quote_shared PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/exports.map")
endif()
if(FQ_BUILD_APPS)
  add_executable(fq_dump apps/fq_dump.cpp)
  target_link_libraries(fq_dump PRIVATE futures_quote_static)
  add_executable(fq_bench apps/fq_bench.cpp)
  target_include_directories(fq_bench PRIVATE tests)
  target_link_libraries(fq_bench PRIVATE futures_quote_static)
  install(TARGETS fq_dump fq_bench RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(FQ_BUILD_TESTS)
  enable_testing()
  add_executable(fq_tests tests/test_main.cpp)
  target_include_directories(fq_tests PRIVATE src)
  target_link_libraries(fq_tests PRIVATE futures_quote_static)
  target_compile_definitions(fq_tests PRIVATE FQ_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME protocols_and_recovery COMMAND fq_tests)
  set_tests_properties(protocols_and_recovery PROPERTIES TIMEOUT 90)
  add_executable(fq_c_example examples/client.c)
  set_target_properties(fq_c_example PROPERTIES LINKER_LANGUAGE CXX)
  target_link_libraries(fq_c_example PRIVATE futures_quote_shared)
  add_test(NAME shared_c_abi COMMAND fq_c_example ${CMAKE_CURRENT_SOURCE_DIR}/config/replay.yaml)
  add_executable(fq_cpp_example examples/client.cpp)
  target_link_libraries(fq_cpp_example PRIVATE futures_quote_static)
  add_test(NAME static_cpp_client COMMAND fq_cpp_example ${CMAKE_CURRENT_SOURCE_DIR}/config/replay.yaml)
  if(FQ_BUILD_APPS)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
      add_test(NAME pcap_end_to_end COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pcap.py $<TARGET_FILE:fq_dump>)
      add_test(NAME schema_check COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_schema_check.py)
    endif()
    add_test(NAME benchmark_smoke COMMAND fq_bench ${CMAKE_CURRENT_SOURCE_DIR}/config/benchmark.yaml 1000)
  endif()
endif()
install(TARGETS futures_quote_shared futures_quote_static EXPORT FuturesQuoteTargets
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/futures_quote DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY config/ DESTINATION ${CMAKE_INSTALL_DATADIR}/futures_quote/config)
install(FILES third_party/yaml-cpp-0.8.0/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/futures_quote/licenses RENAME yaml-cpp-LICENSE)
install(FILES third_party/SBE-LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/futures_quote/licenses)
install(FILES README.md THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/futures_quote)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/futures_quote/docs)
install(EXPORT FuturesQuoteTargets NAMESPACE futures_quote:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/futures_quote)
configure_package_config_file(cmake/futures_quote-config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/futures_quote-config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/futures_quote)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/futures_quote-config-version.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/futures_quote-config.cmake ${CMAKE_CURRENT_BINARY_DIR}/futures_quote-config-version.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/futures_quote)
